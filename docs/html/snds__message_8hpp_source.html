309  static std::string_view extract_token(std::string_view s) noexcept
310  {
311  // If there’s a '\0' anywhere, lose it and everything after.
312  if (auto nz = s.find('\0'); nz != std::string_view::npos)
313  {
314  s.remove_suffix(s.size() - nz);
315  }
316 
317  // Skip leading whitespace (spaces, tabs, CR/LF).
318  auto start = s.find_first_not_of(" \t\r\n");
319  if (start == std::string_view::npos)
320  {
321  return {}; // all whitespace or empty
322  }
323 
324  // Find the end of that first word (up to the next whitespace).
325  auto end = s.find_first_of(" \t\r\n", start);
326  if (end == std::string_view::npos)
327  {
328  end = s.size();
329  }
330 
331  return s.substr(start, end - start);
332  }

705  std::unordered_map<std::string, std::string> unpack_query_params_from_message() const
706  {
707  std::unordered_map<std::string, std::string> params;
708 
709  std::string_view payload = message;
710 
711  // Trim prefix and postfix
712  if (!payload.empty() && payload.front() == parameter_prefix_delimeter) payload.remove_prefix(1);
713  if (!payload.empty() && payload.back() == parameter_postfix_delimeter) payload.remove_suffix(1);
714 
715  // Helper: extract next token
716  auto next_token = [&](std::string_view& sv) -> std::string_view
717  {
718  if (sv.empty()) return {};
719  size_t pos = sv.find(parameter_packing_delimeter);
720  if (pos == std::string_view::npos)
721  {
722  std::string_view out = sv;
723  sv = {};
724  return out;
725  }
726  std::string_view out = sv.substr(0, pos);
727  sv.remove_prefix(pos + 1);
728  return out;
729  };
730 
731  while (!payload.empty())
732  {
733  std::string_view pair_str = next_token(payload);
734  if (pair_str.empty()) continue;
735 
736  // Split key and value
737  size_t eq_pos = pair_str.find(parameter_assignment_delimeter);
738  if (eq_pos == std::string_view::npos)
739  {
740  // Malformed — no '=' found; store empty value
741  params.insert({std::string(pair_str), ""});
742  continue;
743  }
744 
745  std::string_view key = pair_str.substr(0, eq_pos);
746  std::string_view val = pair_str.substr(eq_pos + 1);
747  params.insert({std::string(key), std::string(val)});
748  }
749 
750  return params;
751  }

817  {

820  auto qpos = uri.find('?');
821  std::string_view qs = (qpos == std::string_view::npos) ? uri : uri.substr(qpos + 1);
822  while (!qs.empty() && (qs.front() == '/' || qs.front() == '?')) qs.remove_prefix(1);
823

825  {
826  if (sv.empty()) return {};

828  if (pos == std::string_view::npos)
829  {
830  auto r = sv;
831  sv = {};
832  return r;
833  }
834  auto r = sv.substr(0,pos);
835  sv.remove_prefix(pos+1);
836  return r;
837  };

842  if (pair.empty()) continue;
